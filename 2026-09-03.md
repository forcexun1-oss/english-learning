```python
# importer/adapters/douyin.py
from __future__ import annotations

import re
from datetime import datetime, timezone
from pathlib import Path
from typing import Iterable

import frontmatter

from importer.adapters.base import RawItem
from importer.episode import Episode

_HEADER_RE = re.compile(r"^(#{2,3})\s+(.*)$", re.MULTILINE)


def _split_sections(body: str) -> list[tuple[int, str, str]]:
    """Return ordered (level, title, content) for every ## / ### heading."""
    matches = list(_HEADER_RE.finditer(body))
    sections: list[tuple[int, str, str]] = []
    for i, m in enumerate(matches):
        level = len(m.group(1))
        title = m.group(2).strip()
        start = m.end()
        end = matches[i + 1].start() if i + 1 < len(matches) else len(body)
        content = body[start:end].strip()
        sections.append((level, title, content))
    return sections


def _extract_episode_body(sections: list[tuple[int, str, str]]) -> str | None:
    by_title = {title: content for _, title, content in sections}

    if "要点" in by_title:
        parts = [by_title["要点"]]
        in_yanpan = False
        for level, title, content in sections:
            if title == "研判":
                in_yanpan = True
                continue
            if in_yanpan and level == 2:
                break
            if in_yanpan and title != "关联知识" and content:
                parts.append(content)
        return "\n\n".join(p for p in parts if p)

def _reference_time(metadata: dict, aweme_id: int) -> datetime:
    favorite_at = metadata.get("favorite_at")
    if favorite_at:
        return datetime.fromisoformat(str(favorite_at))
    return datetime.fromtimestamp(aweme_id >> 32, tz=timezone.utc)


class DouyinMarkdownAdapter:
    def __init__(self, notes_dir: Path) -> None:
        self._notes_dir = notes_dir

    def discover(self) -> Iterable[RawItem]:
        for path in sorted(self._notes_dir.glob("*.md")):
            yield RawItem(path=path, mtime=path.stat().st_mtime)

    def to_episode(self, item: RawItem) -> Episode | None:
        post = frontmatter.load(item.path)
        metadata = post.metadata
        aweme_id_raw = metadata.get("aweme_id")
        if not aweme_id_raw:
            return None

        sections = _split_sections(post.content)
        body = _extract_episode_body(sections)
        if not body:
            return None

        aweme_id = int(aweme_id_raw)
        author = metadata.get("author", "")
        title = metadata.get("title", "")

        return Episode(
            name=str(aweme_id),
            episode_body=body,
            source_description=f"抖音收藏 · {author} · {title}",
            reference_time=_reference_time(metadata, aweme_id),
            group_id="douyin",
        )
```

```python
# importer/run.py
from __future__ import annotations

import argparse
import asyncio
import logging
import os
from pathlib import Path

from importer.adapters.base import SourceAdapter
from importer.adapters.douyin import DouyinMarkdownAdapter
from importer.state import ImportState

logger = logging.getLogger("memory_graph.import")


async def run_import(adapter: SourceAdapter, state: ImportState, graphiti) -> tuple[int, int, int]:
    imported = skipped = failed = 0
    await graphiti.build_indices_and_constraints()

    for item in adapter.discover():
        episode = adapter.to_episode(item)
        if episode is None:
            logger.warning("skip unparsable note: %s", item.path)
            skipped += 1
            continue

        if not state.needs_import(episode.name, item.mtime):
            continue

        try:
            await graphiti.add_episode(
                name=episode.name,
                episode_body=episode.episode_body,
                source_description=episode.source_description,
                reference_time=episode.reference_time,
                source=episode.source,
                group_id=episode.group_id,
            )
        except Exception:
            logger.exception("add_episode failed for %s", item.path)
            failed += 1
            continue

        state.mark_imported(episode.name, item.mtime)
        imported += 1

    return imported, skipped, failed


def main() -> None:
    logging.basicConfig(level=logging.INFO)
    parser = argparse.ArgumentParser(prog="memory-graph")
    subparsers = parser.add_subparsers(dest="command", required=True)

    import_parser = subparsers.add_parser("import", help="Import notes into Graphiti")
    import_parser.add_argument("--notes-dir", type=Path, required=True)
    import_parser.add_argument("--db-path", type=Path, default=Path("data/state.db"))

    args = parser.parse_args()
    if args.command != "import":
        return

    from graphiti_core import Graphiti  # deferred: keeps `import importer.run` cheap for tests

    async def _main() -> None:
        adapter = DouyinMarkdownAdapter(args.notes_dir)
        state = ImportState(args.db_path)
        graphiti = Graphiti(
            os.environ["NEO4J_URI"],
            os.environ["NEO4J_USER"],
            os.environ["NEO4J_PASSWORD"],
        )
        try:
            imported, skipped, failed = await run_import(adapter, state, graphiti)
            logger.info("imported=%d skipped=%d failed=%d", imported, skipped, failed)
        finally:
            state.close()
            await graphiti.close()

    asyncio.run(_main())


if __name__ == "__main__":
    main()
```
